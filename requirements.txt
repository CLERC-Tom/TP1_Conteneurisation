# Prérequis pour lancer la partie Kubernetes du TP1 (tâche 3)
# Ce ne sont pas des paquets pip : ce sont les outils à installer sur votre machine.

# ---------------------------------------------------------------
# 1. Docker (obligatoire : kind fait tourner Kubernetes dans Docker)
# ---------------------------------------------------------------
# Linux   : Docker Engine  -> https://docs.docker.com/engine/install/
# Mac/Win : Docker Desktop -> https://docs.docker.com/desktop/
# Vérifier : docker version   (le client ET le serveur doivent répondre)
# Prévoir environ 4 Go de RAM libre pour le cluster.

# ---------------------------------------------------------------
# 2. kubectl (le client qui parle au cluster)
# ---------------------------------------------------------------
# Version : au plus 1 version mineure d'écart avec le cluster (cluster testé : v1.37.0)
# Linux   : curl -LO "https://dl.k8s.io/release/$(curl -L -s https://dl.k8s.io/release/stable.txt)/bin/linux/amd64/kubectl"
#           sudo install -o root -g root -m 0755 kubectl /usr/local/bin/kubectl
# Mac     : brew install kubectl
# Windows : winget install -e --id Kubernetes.kubectl
# Vérifier : kubectl version --client

# ---------------------------------------------------------------
# 3. kind (Kubernetes IN Docker : le cluster local)
# ---------------------------------------------------------------
# Linux   : curl -Lo ./kind https://github.com/kubernetes-sigs/kind/releases/latest/download/kind-linux-amd64
#           chmod +x ./kind && sudo mv ./kind /usr/local/bin/kind
# Mac     : brew install kind
# Windows : winget install Kubernetes.kind
# Vérifier : kind version

# ---------------------------------------------------------------
# Lancer la partie Kubernetes (depuis la racine du dépôt)
# ---------------------------------------------------------------
# kind create cluster --name tp1
# kubectl apply -f k8s/
# kubectl get pods                       # attendre 3 pods en Running
# kubectl port-forward service/serveur-web 8080:80
# -> ouvrir http://localhost:8080
#
# Arrêter : kind delete cluster --name tp1
